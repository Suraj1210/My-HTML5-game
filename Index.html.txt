  <script>
    class SoundEngine {
      constructor() {
        this.ctx = null;
        this.masterGain = null;
        this.sfxEnabled = true;
        this.musicEnabled = true;
        this.masterVolume = 0.8;
        this.sirenActive = false;
        this.sirenOsc1 = null;
        this.sirenOsc2 = null;
        this.sirenGain = null;
        this.sirenInterval = null;
        this.bgMusicTimer = null;
      }

      init() {
        if (!this.ctx) {
          const AudioContext = window.AudioContext || window.webkitAudioContext;
          this.ctx = new AudioContext();
          this.masterGain = this.ctx.createGain();
          this.masterGain.gain.setValueAtTime(this.masterVolume, this.ctx.currentTime);
          this.masterGain.connect(this.ctx.destination);
        }
        if (this.ctx.state === 'suspended') {
          this.ctx.resume();
        }
      }

      setMasterVolume(val) {
        this.masterVolume = Math.max(0, Math.min(1, val));
        if (this.masterGain && this.ctx) {
          this.masterGain.gain.setValueAtTime(this.masterVolume, this.ctx.currentTime);
        }
      }

      playMatchSound(type) {
        if (!this.sfxEnabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.connect(gain);
        gain.connect(this.masterGain);

        if (type === 'solar') {
          osc.type = 'triangle';
          osc.frequency.setValueAtTime(320, now);
          osc.frequency.exponentialRampToValueAtTime(740, now + 0.16);
        } else if (type === 'wind') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(260, now);
          osc.frequency.exponentialRampToValueAtTime(560, now + 0.18);
        } else if (type === 'nuclear') {
          osc.type = 'sawtooth';
          osc.frequency.setValueAtTime(110, now);
          osc.frequency.linearRampToValueAtTime(420, now + 0.22);
        } else if (type === 'cryo') {
          osc.type = 'sine';
          osc.frequency.setValueAtTime(800, now);
          osc.frequency.exponentialRampToValueAtTime(1400, now + 0.25);
        } else if (type === 'battery') {
          osc.type = 'square';
          osc.frequency.setValueAtTime(440, now);
          osc.frequency.linearRampToValueAtTime(880, now + 0.14);
        }

        gain.gain.setValueAtTime(0.24, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.25);
        osc.start(now);
        osc.stop(now + 0.25);
      }

      playChime() {
        if (!this.sfxEnabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const osc = this.ctx.createOscillator();
        const gain = this.ctx.createGain();
        osc.connect(gain);
        gain.connect(this.masterGain);
        osc.type = 'sine';
        osc.frequency.setValueAtTime(587.33, now);
        osc.frequency.exponentialRampToValueAtTime(880, now + 0.12);
        gain.gain.setValueAtTime(0.16, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.15);
        osc.start(now);
        osc.stop(now + 0.15);
      }

      playCryoPurgeSound() {
        if (!this.sfxEnabled || !this.ctx) return;
        const now = this.ctx.currentTime;
        const bufferSize = Math.floor(this.ctx.sampleRate * 0.8);
        const buffer = this.ctx.createBuffer(1, bufferSize, this.ctx.sampleRate);
        const data = buffer.getChannelData(0);
        for (let i = 0; i < bufferSize; i++) {
          data[i] = Math.random() * 2 - 1;
        }
        const noise = this.ctx.createBufferSource();
        noise.buffer = buffer;

        const filter = this.ctx.createBiquadFilter();
        filter.type = 'bandpass';
        filter.frequency.setValueAtTime(1500, now);
        filter.frequency.exponentialRampToValueAtTime(450, now + 0.8);
        filter.Q.setValueAtTime(2.8, now);

        const gain = this.ctx.createGain();
        gain.gain.setValueAtTime(0.3, now);
        gain.gain.exponentialRampToValueAtTime(0.001, now + 0.8);

        noise.connect(filter);
        filter.connect(gain);
        gain.connect(this.masterGain);

        noise.start(now);
        noise.stop(now + 0.8);
      }

      startAirRaidSiren() {
        if (!this.sfxEnabled || !this.ctx || this.sirenActive) return;
        this.sirenActive = true;

        const now = this.ctx.currentTime;
        this.sirenOsc1 = this.ctx.createOscillator();
        this.sirenOsc2 = this.ctx.createOscillator();
        this.sirenGain = this.ctx.createGain();

        this.sirenOsc1.type = 'sawtooth';
        this.sirenOsc2.type = 'triangle';
        this.sirenOsc2.detune.setValueAtTime(15, now);

        this.sirenOsc1.connect(this.sirenGain);
        this.sirenOsc2.connect(this.sirenGain);
        this.sirenGain.connect(this.masterGain);

        this.sirenGain.gain.setValueAtTime(0.01, now);
        this.sirenGain.gain.linearRampToValueAtTime(0.22, now + 0.8);

        this.sirenOsc1.start(now);
        this.sirenOsc2.start(now);

        const runCycle = () => {
          if (!this.sirenActive || !this.ctx) return;
          const t = this.ctx.currentTime;
          this.sirenOsc1.frequency.cancelScheduledValues(t);
          this.sirenOsc2.frequency.cancelScheduledValues(t);
          this.sirenOsc1.frequency.setValueAtTime(320, t);
          this.sirenOsc2.frequency.setValueAtTime(325, t);
          this.sirenOsc1.frequency.exponentialRampToValueAtTime(760, t + 1.5);
          this.sirenOsc2.frequency.exponentialRampToValueAtTime(770, t + 1.5);
          this.sirenOsc1.frequency.exponentialRampToValueAtTime(330, t + 3.0);
          this.sirenOsc2.frequency.exponentialRampToValueAtTime(335, t + 3.0);
        };

        runCycle();
        this.sirenInterval = setInterval(runCycle, 3100);
      }

      stopAirRaidSiren() {
        if (!this.sirenActive) return;
        this.sirenActive = false;
        clearInterval(this.sirenInterval);
        if (this.sirenGain && this.ctx) {
          const now = this.ctx.currentTime;
          this.sirenGain.gain.linearRampToValueAtTime(0.001, now + 0.3);
          setTimeout(() => {
            try {
              if (this.sirenOsc1) this.sirenOsc1.stop();
              if (this.sirenOsc2) this.sirenOsc2.stop();
            } catch(e) {}
          }, 400);
        }
      }

      startMenuMusic() {
        if (!this.musicEnabled || this.bgMusicTimer) return;
        const notes = [130.81, 155.56, 174.61, 196.00, 220.00, 261.63];
        let idx = 0;

        this.bgMusicTimer = setInterval(() => {
          if (!this.musicEnabled || !this.ctx) return;
          const now = this.ctx.currentTime;
          const osc = this.ctx.createOscillator();
          const gain = this.ctx.createGain();
          osc.connect(gain);
          gain.connect(this.masterGain);

          osc.type = 'sine';
          osc.frequency.setValueAtTime(notes[idx % notes.length], now);
          gain.gain.setValueAtTime(0.035, now);
          gain.gain.exponentialRampToValueAtTime(0.0001, now + 0.55);

          osc.start(now);
          osc.stop(now + 0.55);
          idx++;
        }, 320);
      }

      stopMenuMusic() {
        if (this.bgMusicTimer) {
          clearInterval(this.bgMusicTimer);
          this.bgMusicTimer = null;
        }
      }
    }

    function createBoard() {
      boardEl.innerHTML = '';
      grid = [];

      for (let r = 0; r < GRID_SIZE; r++) {
        grid[r] = [];
        for (let c = 0; c < GRID_SIZE; c++) {
          let type;
          do {
            type = getRandomTileType();
          } while (
            (r >= 2 && grid[r - 1][c]?.type === type && grid[r - 2][c]?.type === type) ||
            (c >= 2 && grid[r][c - 1]?.type === type && grid[r][c - 2]?.type === type)
          );

          const cell = { r, c, type, el: null };
          grid[r][c] = cell;
          renderTileDOM(cell);
        }
      }
    }

    function getRandomTileType() {
      const roll = Math.random();
      if (roll < 0.32) return 'solar';
      if (roll < 0.64) return 'wind';
      if (roll < 0.82) return 'nuclear';
      if (roll < 0.92) return 'cryo';
      return 'battery';
    }

    let startX = 0, startY = 0;
    let activeCell = null;
    let isDragging = false;

    function setupPointerGesture(el, cell) {
      el.addEventListener('pointerdown', (e) => {
        sound.init();
        if (isProcessing || !gameActive || isPaused) return;

        startX = e.clientX;
        startY = e.clientY;
        activeCell = cell;
        isDragging = true;
        el.setPointerCapture(e.pointerId);

        if (!selectedTile) {
          selectTile(cell);
        } else {
          if (areNeighbors(selectedTile, cell)) {
            const first = selectedTile;
            clearSelection();
            executeSwap(first, cell);
          } else {
            clearSelection();
            selectTile(cell);
          }
        }
      });

      el.addEventListener('pointermove', (e) => {
        if (!isDragging || activeCell !== cell || !gameActive || isPaused || isProcessing) return;
        const dx = e.clientX - startX;
        const dy = e.clientY - startY;
        const threshold = 18;

        if (Math.abs(dx) > threshold || Math.abs(dy) > threshold) {
          isDragging = false;
          let targetR = cell.r;
          let targetC = cell.c;

          if (Math.abs(dx) > Math.abs(dy)) {
            targetC += dx > 0 ? 1 : -1;
          } else {
            targetR += dy > 0 ? 1 : -1;
          }

          if (targetR >= 0 && targetR < GRID_SIZE && targetC >= 0 && targetC < GRID_SIZE) {
            const targetCell = grid[targetR][targetC];
            clearSelection();
            executeSwap(cell, targetCell);
          }
        }
      });

      el.addEventListener('pointerup', () => { isDragging = false; });
      el.addEventListener('pointercancel', () => { isDragging = false; });
    }

    function areNeighbors(a, b) {
      return (Math.abs(a.r - b.r) + Math.abs(a.c - b.c)) === 1;
    }

    async function executeSwap(cellA, cellB) {
      if (isProcessing) return;
      isProcessing = true;

      swapGridCells(cellA, cellB);
      await new Promise(r => setTimeout(r, 160));

      const matches = findMatches();
      if (matches.length > 0) {
        await resolveMatches(matches);
      } else {
        swapGridCells(cellA, cellB);
        await new Promise(r => setTimeout(r, 160));
      }

      isProcessing = false;
    }

    function swapGridCells(a, b) {
      const tempType = a.type;
      a.type = b.type;
      b.type = tempType;

      refreshTileVisual(a);
      refreshTileVisual(b);
    }

    function refreshTileVisual(cell) {

    function findMatches() {
      const matched = new Set();

      for (let r = 0; r < GRID_SIZE; r++) {
        for (let c = 0; c < GRID_SIZE - 2; c++) {
          const type = grid[r][c].type;
          if (type && grid[r][c + 1].type === type && grid[r][c + 2].type === type) {
            matched.add(grid[r][c]);
            matched.add(grid[r][c + 1]);
            matched.add(grid[r][c + 2]);
          }
        }
      }

      for (let c = 0; c < GRID_SIZE; c++) {
        for (let r = 0; r < GRID_SIZE - 2; r++) {
          const type = grid[r][c].type;
          if (type && grid[r + 1][c].type === type && grid[r + 2][c].type === type) {
            matched.add(grid[r][c]);
            matched.add(grid[r + 1][c]);
            matched.add(grid[r + 2][c]);
          }
        }
      }

      return Array.from(matched);
    }

    async function resolveMatches(matchedCells) {
      let wattsEarned = 0;
      let coolEarned = 0;

      matchedCells.forEach(cell => {
        const info = TILE_TYPES[cell.type];
        wattsEarned += info.watt;
        if (info.cool) coolEarned += info.cool;

        sound.playMatchSound(cell.type);
        triggerTileBurst(cell);
        spawnFloatingEnergyOrb(cell, info.color);
      });

      currentPower += wattsEarned;
      coreTemp = Math.max(28, coreTemp - coolEarned);
      updateHUD();

      await new Promise(r => setTimeout(r, 260));

      matchedCells.forEach(cell => {
        cell.type = null;
        cell.el.innerHTML = '';
        cell.el.className = 'tile';
      });

      for (let c = 0; c < GRID_SIZE; c++) {
        let emptyRow = GRID_SIZE - 1;
        for (let r = GRID_SIZE - 1; r >= 0; r--) {
          if (grid[r][c].type !== null) {
            if (emptyRow !== r) {
              grid[emptyRow][c].type = grid[r][c].type;
              grid[r][c].type = null;
              refreshTileVisual(grid[emptyRow][c]);
              grid[r][c].el.innerHTML = '';
            }
            emptyRow--;
          }
        }
        for (let r = emptyRow; r >= 0; r--) {
          grid[r][c].type = getRandomTileType();
          refreshTileVisual(grid[r][c]);
        }
      }

      await new Promise(r => setTimeout(r, 180));

      const cascade = findMatches();
      if (cascade.length > 0) {
        await resolveMatches(cascade);
      } else {
        checkWinCondition();
      }
    }

    function triggerTileBurst(cell) {
      cell.el.classList.add('tile-burst');
      const rect = cell.el.getBoundingClientRect();
      const cx = rect.left + rect.width / 2;
      const cy = rect.top + rect.height / 2;

      for (let i = 0; i < 7; i++) {
        particles.push({
          x: cx,
          y: cy,
          vx: (Math.random() - 0.5) * 5,
          vy: (Math.random() - 0.5) * 5,
          radius: Math.random() * 2 + 1,
          color: TILE_TYPES[cell.type]?.color || '#00ff66',
          alpha: 1,
          decay: 0.04
        });
      }
    }

    function spawnFloatingEnergyOrb(cell, color) {
      const rect = cell.el.getBoundingClientRect();
      const targetRect = powerProgressBar.getBoundingClientRect();

      const startX = rect.left + rect.width / 2;
      const startY = rect.top + rect.height / 2;
      const endX = targetRect.left + (targetRect.width * Math.min(1, currentPower / powerTarget));
      const endY = targetRect.top + targetRect.height / 2;

      particles.push({
        isProjectile: true,
        startX,
        startY,
        endX,
        endY,
        progress: 0,
        speed: 0.045 + Math.random() * 0.02,
        cpX: (startX + endX) / 2 + (Math.random() - 0.5) * 80,
        cpY: Math.min(startY, endY) - 40,
        color: color || '#00ff66'
      });
    }

    function renderFX() {
      fxCtx.clearRect(0, 0, fxCanvas.width, fxCanvas.height);

      for (let i = particles.length - 1; i >= 0; i--) {
        const p = particles[i];

        if (p.isProjectile) {
          p.progress += p.speed;
          if (p.progress >= 1) {
            sound.playChime();
            particles.splice(i, 1);
            continue;
          }
          const t = p.progress;
          const curX = (1 - t) * (1 - t) * p.startX + 2 * (1 - t) * t * p.cpX + t * t * p.endX;
          const curY = (1 - t) * (1 - t) * p.startY + 2 * (1 - t) * t * p.cpY + t * t * p.endY;

          fxCtx.beginPath();
          fxCtx.arc(curX, curY, 3.5, 0, Math.PI * 2);
          fxCtx.fillStyle = p.color;
          fxCtx.shadowColor = p.color;
          fxCtx.shadowBlur = 6;
          fxCtx.fill();
          fxCtx.shadowBlur = 0;
        } else {
          p.x += p.vx;
          p.y += p.vy;
          p.alpha -= p.decay;

          if (p.alpha <= 0) {
            particles.splice(i, 1);
            continue;
          }

          fxCtx.beginPath();
          fxCtx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);
          fxCtx.fillStyle = p.color;
          fxCtx.globalAlpha = p.alpha;
          fxCtx.fill();
          fxCtx.globalAlpha = 1;
        }
      }

      requestAnimationFrame(renderFX);
    }
    requestAnimationFrame(renderFX);

    function startFacilityHeartbeat() {
      if (tickerInterval) clearInterval(tickerInterval);

      tickerInterval = setInterval(() => {
        if (!gameActive || isPaused) return;

        coreTemp += 2.0;

        if (coreTemp > 65) {
          civicWater = Math.max(0, civicWater - 2.4);
        }

        if (coreTemp > 80) {
          civicGrid = Math.max(0, civicGrid - 2.8);
        }

        if (coreTemp >= 75) {
          sound.startAirRaidSiren();
          tempGaugeCard.classList.add('alert-siren-active');
        } else {
          sound.stopAirRaidSiren();
          tempGaugeCard.classList.remove('alert-siren-active');
        }

        if (civicWater < 50 && !hasShownWaterWarningThisLevel) {
          triggerWaterProtestModal();
        }

        if (coreTemp >= 100) {
          triggerGameOver('meltdown');
        } else if (civicWater <= 0) {
          triggerGameOver('water');
        } else if (civicGrid <= 0) {
          triggerGameOver('blackout');
        }

        updateHUD();
      }, 1000);
    }

    function updateHUD() {
      const pct = Math.min(100, Math.round((currentPower / powerTarget) * 100));
      powerProgressBar.style.width = `${pct}%`;
      powerPercentLabel.innerText = `${pct}% QUOTA`;
      powerRatioText.innerText = `${currentPower}W / ${powerTarget}W`;

      if (pct >= 100) {
        powerStatus.innerText = 'STATUS: COMPLETE';
        powerStatus.className = 'text-green-400 font-bold';
      } else {
        powerStatus.innerText = 'STATUS: DEFICIT';
        powerStatus.className = 'text-amber-400 font-semibold';
      }

      tempValue.innerText = `${Math.round(coreTemp)}°C`;
      tempBar.style.width = `${Math.min(100, coreTemp)}%`;

      if (coreTemp >= 75) {
        tempBar.className = 'h-full bg-red-500 transition-all duration-300';
        tempValue.className = 'font-black text-red-400 font-heading text-xs animate-pulse';
        tempAlertLabel.innerText = 'KLAXON ALARM';
        tempAlertLabel.className = 'text-[9px] text-red-400 font-bold truncate mt-0.5';
      } else if (coreTemp >= 55) {
        tempBar.className = 'h-full bg-amber-500 transition-all duration-300';
        tempValue.className = 'font-bold text-amber-400 font-heading text-xs';
        tempAlertLabel.innerText = 'ELEVATED HEAT';
        tempAlertLabel.className = 'text-[9px] text-amber-400 font-medium truncate mt-0.5';
      } else {
        tempBar.className = 'h-full bg-green-500 transition-all duration-300';
        tempValue.className = 'font-bold text-green-400 font-heading text-xs';
        tempAlertLabel.innerText = 'NOMINAL';
        tempAlertLabel.className = 'text-[9px] text-green-400/90 font-semibold truncate mt-0.5';
      }

      waterValue.innerText = `${Math.round(civicWater)}%`;
      waterBar.style.width = `${civicWater}%`;
      waterBar.className = civicWater < 30 ? 'h-full bg-red-500' : 'h-full bg-blue-500';

      gridValue.innerText = `${Math.round(civicGrid)}%`;
      gridBar.style.width = `${civicGrid}%`;
      gridBar.className = civicGrid < 30 ? 'h-full bg-red-500' : 'h-full bg-amber-500';
    }

    function triggerWaterProtestModal() {
      hasShownWaterWarningThisLevel = true;
      const ev = CRISIS_EVENTS[Math.floor(Math.random() * CRISIS_EVENTS.length)];
      crisisHeadline.innerText = ev.headline;
      crisisReport.innerText = ev.story;
      crisisImage.src = ev.image;
      crisisModal.classList.remove('hidden');
    }

    function checkWinCondition() {
      if (currentPower >= powerTarget && gameActive) {
        gameActive = false;
        sound.stopAirRaidSiren();

        if (level === 10) {
          evaluateLevel10GrandFinale();
        } else {
          showVictoryModal();
        }
      }
    }

    /* GUARANTEED WINNING NEWS ARTICLE IMAGE DISPLAY */
    function showVictoryModal() {
      const story = VICTORY_STORIES_ALL_LEVELS[level - 1];
      victoryHeadline.innerText = story.headline;
      victoryStory.innerText = story.story;
      victorySource.innerText = story.source;

      // Set primary image with automatic dynamic fallback
      const fallback = generateFallbackGraphic(story.fallbackTitle, story.fallbackSubtitle);
      victoryNewsImg.onerror = function() {
        this.onerror = null;
        this.src = fallback;
      };
      victoryNewsImg.src = story.image || fallback;

      statWater.innerText = `${Math.round(civicWater)}% INTAKE`;
      statGrid.innerText = `${Math.round(civicGrid)}% RESERVE`;

      victoryModal.classList.remove('hidden');
    }

    function triggerGameOver(reason) {
      if (!gameActive) return;
      gameActive = false;
      sound.stopAirRaidSiren();

      const defeat = DEFEAT_STORIES[reason];
      defeatHeadline.innerText = defeat.headline;
      defeatStory.innerText = defeat.story;
      defeatNewsImg.src = defeat.image;
      defeatSource.innerText = defeat.source;

      defeatModal.classList.remove('hidden');
    }

    function openTutorialModal() {
      switchTutorialTab(1);
      tutorialModal.classList.remove('hidden');
    }

    function switchTutorialTab(tabNum) {
      [tutTab1, tutTab2, tutTab3, tutTab4].forEach(tab => tab.classList.add('hidden'));
      [tutTab1Btn, tutTab2Btn, tutTab3Btn, tutTab4Btn].forEach(btn => {
        btn.className = 'text-[10px] font-bold px-2.5 py-1 rounded-lg bg-green-950 text-green-300 border border-green-800 whitespace-nowrap';
      });

      if (tabNum === 1) {
        tutTab1.classList.remove('hidden');
        tutTab1Btn.className = 'text-[10px] font-bold px-2.5 py-1 rounded-lg bg-green-500 text-black whitespace-nowrap';
      } else if (tabNum === 2) {
        tutTab2.classList.remove('hidden');
        tutTab2Btn.className = 'text-[10px] font-bold px-2.5 py-1 rounded-lg bg-green-500 text-black whitespace-nowrap';
      } else if (tabNum === 3) {
        tutTab3.classList.remove('hidden');
        tutTab3Btn.className = 'text-[10px] font-bold px-2.5 py-1 rounded-lg bg-green-500 text-black whitespace-nowrap';
      } else if (tabNum === 4) {
        tutTab4.classList.remove('hidden');
        tutTab4Btn.className = 'text-[10px] font-bold px-2.5 py-1 rounded-lg bg-green-500 text-black whitespace-nowrap';
      }
    }

    tutTab3Btn.addEventListener('click', () => switchTutorialTab(3));
    tutTab4Btn.addEventListener('click', () => switchTutorialTab(4));

    tutCloseBtn.addEventListener('click', () => {
      tutorialModal.classList.add('hidden');

    function togglePause() {
      if (!gameActive) return;
      isPaused = !isPaused;
      if (isPaused) {
        sound.stopAirRaidSiren();
        pauseModal.classList.remove('hidden');
      } else {
        pauseModal.classList.add('hidden');
      }
    }

      if (e.key === 'Escape') togglePause();
    });
    pauseBtn.addEventListener('click', togglePause);
